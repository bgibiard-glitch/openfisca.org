{{- /* event-dates — a localized event date range built from start/end dates. Usage: {{ partial "event-dates.html" (dict "start" .Params.start_date "end" .Params.end_date) }} Month names are localized to the current site language (EN/FR) via time.Format; days carry EN ordinals (30th) / FR "1er". Adapts to same day, same month, cross-month (same year) and cross-year ranges. */ -}} {{- $start := time.AsTime .start -}} {{- $end := time.AsTime .end -}} {{- $lang := site.Language.Lang -}} {{- $sd := partial "event-day.html" (dict "day" $start.Day "lang" $lang) -}} {{- $ed := partial "event-day.html" (dict "day" $end.Day "lang" $lang) -}} {{- $out := "" -}} {{- if eq ($start.Format "2006-01-02") ($end.Format "2006-01-02") -}} {{- /* single day — e.g. 30th March 2026 */ -}} {{- $out = printf "%s %s" $sd (time.Format "January 2006" $start) -}} {{- else if eq ($start.Format "2006-01") ($end.Format "2006-01") -}} {{- /* same month — e.g. 30th–31st March 2026 */ -}} {{- $out = printf "%s–%s %s" $sd $ed (time.Format "January 2006" $start) -}} {{- else if eq ($start.Format "2006") ($end.Format "2006") -}} {{- /* cross-month, same year — e.g. 30th April – 2nd May 2026 */ -}} {{- $out = printf "%s %s – %s %s" $sd (time.Format "January" $start) $ed (time.Format "January 2006" $end) -}} {{- else -}} {{- /* cross-year — e.g. 31st December 2026 – 2nd January 2027 */ -}} {{- $out = printf "%s %s – %s %s" $sd (time.Format "January 2006" $start) $ed (time.Format "January 2006" $end) -}} {{- end -}} {{- return ($out | safeHTML) -}}
